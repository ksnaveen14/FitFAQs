# FitRAG — requirements.txt
# Install with: pip install -r requirements.txt
#
# Pinning strategy: exact patch pins, not floors ("==" not ">=").
# Rationale: ragas jumped 0.1.x -> 0.4.x with breaking API changes (metric
# imports, evaluator signatures). An unbounded ">=" would let a future
# `pip install` silently pull a newer major and break Eval without warning.

# --- Core RAG stack ---
openai==1.51.2                 # your H4 generation code targets this. NOTE: ragas below pulls langchain-openai, which will transitively resolve openai to ~1.109.x in your actual env — that's fine, langchain-openai 0.1.25 supports that range. Don't fight it with a hard pin conflict.
faiss-cpu==1.9.0
sentence-transformers==3.1.1   # latest 3.x line; skip 4.x/5.x/6.x (current latest) until re-tested against H2 embedding code
rank-bm25==0.2.2                # this IS latest — project is unmaintained (4 releases ever, years since last). No newer version exists; accepted risk, not a bug you can fix by bumping.

# --- Evaluation ---
# ragas 0.2.15 declares langchain-community with NO version ceiling.
# Without the pins below, pip installs whatever langchain-community is
# current today (0.4.x+), which has restructured/removed the
# langchain_community.chat_models.vertexai import path ragas 0.2.x's
# import chain expects at load time -> ModuleNotFoundError on `from ragas import evaluate`.
# These four must be installed together, same generation as ragas 0.2.15.
ragas==0.2.15                   # latest 0.2.x. Deliberately NOT 0.4.x (current latest) — 0.3/0.4 rewrote metric imports + evaluator signatures. Upgrading is a migration task, not a version bump.
langchain==0.2.17
langchain-core==0.2.43
langchain-community==0.2.19
langchain_openai==0.1.25
datasets==2.21.0

# --- Document loading ---

# --- Frontend ---
streamlit==1.39.0

# --- Utilities ---
numpy==1.26.4                   # capped below 2.x deliberately — faiss-cpu/sentence-transformers wheels above target numpy 1.x ABI
httpx==0.27.2                   # openai==1.51.2 declares "httpx<1,>=0.23.0" (no real ceiling), but its internal client still passes `proxies=` to httpx.Client(), which httpx 0.28 removed. Verified failure: TypeError: Client.__init__() got an unexpected keyword argument 'proxies'. Pin below 0.28 until upgrading openai SDK deliberately.
